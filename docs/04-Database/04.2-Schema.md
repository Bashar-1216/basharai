## 1. Prisma Schema Definition (Source of Truth)

يمثل كود **Prisma Schema** التالي العقد البرمجي الموحد والمصدر الوحيد للحقيقة (Single Source of Truth) لقاعدة بيانات PostgreSQL الحية، ويتم توليد حزم العميل (Typescript types) منه مباشرة.

```prisma
// src/frontend/prisma/schema.prisma

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

// ==========================================
// 1. Auth Domain (NextAuth.js v5 compatible)
// ==========================================

model User {
  id            String    @id @default(uuid()) @db.Uuid
  name          String?
  email         String?   @unique
  emailVerified DateTime?
  image         String?
  role          String    @default("VISITOR") // ADMIN or VISITOR
  createdAt     DateTime  @default(now())
  accounts      Account[]
  sessions      Session[]
}

model Account {
  id                String  @id @default(uuid()) @db.Uuid
  userId            String  @db.Uuid
  type              String
  provider          String
  providerAccountId String
  refresh_token     String? @db.Text
  access_token      String? @db.Text
  expires_at        Int?
  token_type        String?
  scope             String?
  id_token          String? @db.Text
  session_state     String?

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerAccountId])
}

model Session {
  id           String   @id @default(uuid()) @db.Uuid
  sessionToken String   @unique
  userId       String   @db.Uuid
  expires      DateTime
  user         User     @relation(fields: [userId], references: [id], onDelete: Cascade)
}

// ==========================================
// 2. Content Domain (Bilingual Portfolio)
// ==========================================

model Project {
  id             String    @id @default(uuid()) @db.Uuid
  slug           String    @unique
  titleEn        String    @map("title_en")
  titleAr        String    @map("title_ar")
  descriptionEn  String    @db.Text @map("description_en")
  descriptionAr  String    @db.Text @map("description_ar")
  githubUrl      String?   @map("github_url")
  liveUrl        String?   @map("live_url")
  publishedAt    DateTime  @default(now()) @map("published_at")
  featured       Boolean   @default(false)
}

model Experience {
  id          String    @id @default(uuid()) @db.Uuid
  company     String    @unique
  logoUrl     String?   @map("logo_url")
  titleEn     String    @map("title_en")
  titleAr     String    @map("title_ar")
  startDate   DateTime  @map("start_date")
  endDate     DateTime? @map("end_date")
  summaryEn   String    @db.Text @map("summary_en")
  summaryAr   String    @db.Text @map("summary_ar")
  isCurrent   Boolean   @default(false) @map("is_current")
}

model BlogPost {
  id          String   @id @default(uuid()) @db.Uuid
  slug        String   @unique
  titleEn     String   @map("title_en")
  titleAr     String   @map("title_ar")
  contentEn   String   @db.Text @map("content_en")
  contentAr   String   @db.Text @map("content_ar")
  publishedAt DateTime @default(now()) @map("published_at")
  published   Boolean  @default(false)
}

// ==========================================
// 3. AI / RAG Domain
// ==========================================

model Document {
  id         String   @id @default(uuid()) @db.Uuid
  sourceUrl  String   @map("source_url")
  titleEn    String   @map("title_en")
  titleAr    String   @map("title_ar")
  docType    String   @default("PROJECT") @map("doc_type") // EXPERIENCE, PROJECT, BLOG
  fileHash   String   @map("file_hash")
  updatedAt  DateTime @updatedAt @map("updated_at")
  chunks     Chunk[]
}

// Note: Vector fields are created via custom migration SQL (pgvector)
model Chunk {
  id          String   @id @default(uuid()) @db.Uuid
  documentId  String   @db.Uuid @map("document_id")
  contentEn   String   @db.Text @map("content_en")
  contentAr   String   @db.Text @map("content_ar")
  chunkHash   String   @map("chunk_hash")
  
  document    Document @relation(fields: [documentId], references: [id], onDelete: Cascade)
}

model Conversation {
  id           String    @id @default(uuid()) @db.Uuid
  sessionId    String    @unique @map("session_id") // Fingerprint SHA-256
  userLanguage String    @default("en") @map("user_language")
  createdAt    DateTime  @default(now()) @map("createdAt")
  updatedAt    DateTime  @updatedAt @map("updatedAt")
  messages     Message[]
  llmLogs      LLMLog[]
}

model Message {
  id             String       @id @default(uuid()) @db.Uuid
  conversationId String       @db.Uuid @map("conversation_id")
  role           String       // USER or ASSISTANT
  content        String       @db.Text
  createdAt      DateTime     @default(now()) @map("created_at")
  
  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
}

// ==========================================
// 4. Observability & Eval Domain
// ==========================================

model LLMLog {
  id             String       @id @default(uuid()) @db.Uuid
  conversationId String       @db.Uuid @map("conversation_id")
  queryText      String       @db.Text @map("query_text")
  responseText   String       @db.Text @map("response_text")
  tokensInput    Int          @map("tokens_input")
  tokensOutput   Int          @map("tokens_output")
  costUsd        Float        @map("cost_usd")
  latencyMs      Int          @map("latency_ms")
  languageCode   String       @default("en") @map("language_code")
  createdAt      DateTime     @default(now()) @map("created_at")

  conversation   Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
  evaluations    EvaluationResult[]
}

model EvaluationResult {
  id             String   @id @default(uuid()) @db.Uuid
  llmLogId       String   @db.Uuid @map("llm_log_id")
  runId          String   @map("run_id") // UUID of evaluation batch
  contextRelevance Float  @map("context_relevance") // 0.0 to 1.0
  groundedness     Float  @map("groundedness") // 0.0 to 1.0
  answerRelevance  Float  @map("answer_relevance") // 0.0 to 1.0
  passed         Boolean  @default(true)
  feedbackReason String?  @db.Text @map("feedback_reason")
  createdAt      DateTime @default(now()) @map("created_at")

  llmLog         LLMLog   @relation(fields: [llmLogId], references: [id], onDelete: Cascade)
}
```

## 2. pgvector & Custom SQL Additions

لا تدعم أداة Prisma تعريف الأبعاد المتجهية (Vector) أو بناء فهارس HNSW الخاصة بـ pgvector بشكل أصيل حتى الآن. لذلك، يتم كتابتها كـ **ملفات هجرة SQL مخصصة (Custom Migration SQL)** يتم تشغيلها بالتوازي:

```sql
-- migration.sql
-- 1. Enable the vector extension in PostgreSQL
CREATE EXTENSION IF NOT EXISTS vector;

-- 2. Alter Chunk table to add English and Arabic vector fields (1536 dimensions)
ALTER TABLE "Chunk" ADD COLUMN "embedding_en" vector(1536);
ALTER TABLE "Chunk" ADD COLUMN "embedding_ar" vector(1536);

-- 3. Create HNSW indexes using Cosine Distance metric for faster retrieval
CREATE INDEX IF NOT EXISTS "Chunk_embedding_en_hnsw_idx" 
ON "Chunk" USING hnsw (embedding_en vector_cosine_ops) 
WITH (m = 16, ef_construction = 64);

CREATE INDEX IF NOT EXISTS "Chunk_embedding_ar_hnsw_idx" 
ON "Chunk" USING hnsw (embedding_ar vector_cosine_ops) 
WITH (m = 16, ef_construction = 64);
```

## 3. Index Strategy & Constraints | استراتيجية الفهارس والقيود

لتسريع استعلامات تصفح الزوار والتحليلات ومنع الازدواجية:

1. **معرفات المسار الفريدة (Unique Slugs):**
   - الحقول `slug` في جدولي `Project` و `BlogPost` معرّفة بـ `@unique` لضمان عدم ازدواجية الروابط وبناء فهارس افتراضية سريعة للاستعلام بـ URL.
2. **معرفات الجلسة (Session Identifiers):**
   - الحقل `sessionId` في جدول `Conversation` معرّف كـ `@unique` لمنع فتح أكثر من سجل محادثة نشط لنفس البصمة الرقمية للزائر.
3. **مفتاح OAuth الفريد (OAuth Constraints):**
   - القيد `@@unique([provider, providerAccountId])` يمنع ربط نفس حساب Google الخارجي بمسؤولين متعددين في قاعدة البيانات.

## 4. Bilingual Parity Implementation Patterns

لضمان متطلبات i18n البرمجية من العمارة 02.4:
- قمنا بفصل العناوين والأوصاف إلى حقول صريحة: `titleEn` / `titleAr` و `descriptionEn` / `descriptionAr`.
- هذا يضمن دقة استرجاع البيانات للـ API المناسب بناءً على لغة مسار المستخدم (`/en` أو `/ar`) دون استهلاك موارد المعالجة للترجمة الديناميكية.
